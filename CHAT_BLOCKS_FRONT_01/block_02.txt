function extractAssistantText(data: unknown): string {
  if (!data || typeof data !== "object") {
    return "Je n’ai pas pu produire de réponse exploitable pour le moment.";
  }

  const record = data as Record<string, unknown>;

  const directKeys = ["answer", "response", "content", "message", "text"];
  for (const key of directKeys) {
    const value = record[key];
    if (typeof value === "string" && value.trim()) {
      return value.trim();
    }
  }

  const choices = record.choices;
  if (Array.isArray(choices) && choices.length > 0) {
    const first = choices[0] as Record<string, unknown>;
    const message = first.message as Record<string, unknown> | undefined;
    const content = message?.content;
    if (typeof content === "string" && content.trim()) {
      return content.trim();
    }
  }

  return "Je n’ai pas pu produire de réponse exploitable pour le moment.";
}

export default function AssistantPage() {
  const [messages, setMessages] = useState<MessageItem[]>([

      role: "assistant",
      content:
        "Bonjour. Je suis CloneChat. Je sers à expliquer CloneStore, orienter vers la bonne zone du produit et réduire les frictions dans l’usage.",
    },
  ]);
  const [input, setInput] = useState("");
  const [loading, setLoading] = useState(false);

  const canSend = useMemo(() => input.trim().length > 0 && !loading, [input, loading]);

  async function sendMessage(rawText: string) {
    const clean = rawText.trim();
    if (!clean || loading) return;

    setMessages((prev) => [...prev, { role: "user", content: clean }]);
    setInput("");
    setLoading(true);

    try {
      const response = await fetch("/api/assistant", {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({
          message: clean,
          input: clean,
          query: clean,
        }),
      });

      const data = await response.json().catch(() => null);
      const answer = extractAssistantText(data);

      setMessages((prev) => [...prev, { role: "assistant", content: answer }]);
    } catch {
      setMessages((prev) => [
        ...prev,
        {
          role: "assistant",
          content:
            "Je n’ai pas pu répondre correctement pour le moment. Tu peux continuer via la Boutique, Mes employés ou Mon espace.",

            <div className="space-y-4">

                {agent.short}

              {agent.slug === "pierre" ? (

                  <Button asChild>
                    <Link href="/profile">

                      <ArrowRight className="h-4 w-4" />

              ))}
            </div>

            {agent.note ? (
              <div className="rounded-[1.2rem] border border-[color:var(--cs-line-soft)] bg-white/52 px-4 py-3">
                <p className="text-xs uppercase tracking-[0.16em] text-[color:var(--cs-ink-4)]">

                <Clock3 className="mt-0.5 h-4 w-4 text-[color:var(--cs-violet)]" />
                <p className="text-sm text-[color:var(--cs-ink-3)]">
                  {agent.slug === "pierre"
                    ? "L’accès commercial reste volontairement fermé tant que l’ouverture n’est pas complètement prête."
                    : "Cette fiche existe pour poser la vision du système sans prétendre à une ouverture immédiate."}
                </p>
              </div>

          </div>
        </div>

      <section className="grid gap-4 md:grid-cols-3">
        {agent.surfaces.map((item) => (
          <div key={item.title} className="cs-card h-full">
            <div className="space-y-3">
              <div className="flex items-center gap-2 text-[color:var(--cs-violet)]">
                {item.title === "Traçabilité" ? (
                  <ShieldCheck className="h-4 w-4" />
                ) : item.title === "Mission libre" || item.title === "Interne" ? (
                  <Waypoints className="h-4 w-4" />
                ) : (
                  <Sparkles className="h-4 w-4" />
                )}
                <span className="text-sm font-medium text-[color:var(--cs-ink-1)]">
                  {item.title}
                </span>
              </div>

              <p className="text-sm leading-6 text-[color:var(--cs-ink-3)]">
                {item.text}
              </p>
            </div>
          </div>
        ))}
      </section>
    </div>
  );
