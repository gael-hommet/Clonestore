function fmtDate(value: string | null | undefined) {
  if (!value) return "—";
  const d = new Date(value);
  if (Number.isNaN(d.getTime())) return value;
  return d.toLocaleDateString("fr-FR", {
    year: "numeric",
    month: "2-digit",
    day: "2-digit",
  });
}

function statusClass(status: string) {
  const st = status.toLowerCase();

  if (st === "active") return "cs-status cs-status--success";
  if (st === "trialing") return "cs-status cs-status--info";
  if (st === "past_due") return "cs-status cs-status--warn";
  if (st === "cancelled") return "cs-status";
  return "cs-status";
}

function statusLabel(status: string) {
  const st = status.toLowerCase();

  if (st === "active") return "Actif";
  if (st === "trialing") return "Essai";
  if (st === "past_due") return "Paiement en attente";
  if (st === "cancelled") return "Résilié";
  return status || "Inconnu";
}

function getAgentMeta(slug: string): AgentMeta {
  return (
    AGENT_META[slug] ?? {
      name: slug.charAt(0).toUpperCase() + slug.slice(1),
      role: "Employé IA CloneStore",
      description:
        "Employé IA rattaché à l’écosystème CloneStore.",
      statusLabelWhenMissing: "Bientôt",
    }
  );
}

  label,
  value,
  helper,
}: {
  label: string;
  value: string;
  helper: string;
}) {
  return (
    <div className="cs-card">
      <div className="relative space-y-2">
        <p className="text-xs uppercase tracking-[0.16em] text-[var(--cs-ink-4)]">
          {label}
        </p>
        <p className="text-2xl font-semibold tracking-[-0.03em] text-[var(--cs-ink-1)]">
          {value}
        </p>
        <p className="text-xs text-[var(--cs-ink-4)]">{helper}</p>

export default function Client() {
  const [orders, setOrders] = useState<OrderRow[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);
  const [filter, setFilter] = useState<"all" | "active" | "catalog">("all");

  const supabase = useMemo(() => getSupabase() as SupabaseClient | null, []);

  useEffect(() => {
    let active = true;

    async function load() {
      setLoading(true);
      setError(null);

      if (!supabase) {
        if (!active) return;
        setError(
          "Configuration Supabase manquante. Vérifie les variables d’environnement et redémarre le projet.",
        );
        setLoading(false);
        return;
      }

      try {
        const { data: userRes, error: userErr } = await supabase.auth.getUser();

        if (userErr) throw userErr;

        const user = userRes.user;
        if (!user) {
          if (!active) return;
          setOrders([]);
          setLoading(false);
          return;
        }

        const { data, error: ordersErr } = await supabase
          .from("orders")
          .select("id, agent_slug, status, started_at, ended_at")
          .eq("user_id", user.id)
          .order("started_at", { ascending: false });

        if (ordersErr) throw ordersErr;

        if (!active) return;
        setOrders((data ?? []) as OrderRow[]);
        setLoading(false);
      } catch (err) {
        if (!active) return;
        setError(
          err instanceof Error
            ? err.message
            : "Impossible de charger les employés pour le moment.",
        );
        setLoading(false);
      }
    }

    load();

    return () => {
      active = false;
    };
  }, [supabase]);

  const ordersBySlug = useMemo(() => {
    const map = new Map<string, OrderRow>();

    for (const order of orders) {
      if (!map.has(order.agent_slug)) {
        map.set(order.agent_slug, order);
      }
    }

    return map;
  }, [orders]);

  const activeOrders = useMemo(
    () => orders.filter((o) => o.status.toLowerCase() === "active"),
    [orders],
  );

  const visibleSlugs = useMemo(() => {
    const set = new Set<string>(CATALOG_ORDER);

    for (const order of orders) {
      set.add(order.agent_slug);
    }

    const base = Array.from(set);

    if (filter === "active") {
      return base.filter((slug) => {
        const order = ordersBySlug.get(slug);
        return order?.status.toLowerCase() === "active";
      });
    }

    if (filter === "catalog") {
      return CATALOG_ORDER;
    }

    return base;
  }, [filter, orders, ordersBySlug]);

  return (
    <div className="space-y-6">
      <section className="grid gap-4 md:grid-cols-3">
        <StatTile
          label="Employés actifs"
          value={String(activeOrders.length)}
          helper="Accès réellement utilisables aujourd’hui."
        />
        <StatTile
          label="Historique lié au compte"
          value={String(orders.length)}
          helper="Employés déjà activés ou passés par le compte."
        />
        <StatTile
          label="Catalogue visible"
          value={String(CATALOG_ORDER.length)}
          helper="Présence produit globale CloneStore."
        />
      </section>

      <section className="cs-card">
        <div className="relative flex flex-col gap-4 lg:flex-row lg:items-center lg:justify-between">
          <div className="space-y-1">
            <p className="text-sm font-semibold text-[var(--cs-ink-1)]">
              Filtrer la vue
            </p>
            <p className="text-sm text-[var(--cs-ink-4)]">
              Pour quelqu’un de non technique, la navigation doit rester simple
              et instantanée.
            </p>
          </div>

          <div className="flex flex-wrap gap-2">
            <button
              type="button"
              onClick={() => setFilter("all")}
              className={[
                "rounded-full border px-4 py-2 text-sm font-medium transition",
                filter === "all"
                  ? "border-[color:color-mix(in_srgb,var(--cs-account-accent)_22%,white)] bg-[color:color-mix(in_srgb,var(--cs-account-accent)_12%,white)] text-[var(--cs-ink-1)]"
                  : "border-[color:color-mix(in_srgb,var(--cs-line-soft)_68%,white)] bg-white/52 text-[var(--cs-ink-3)]",
              ].join(" ")}
            >
              Tout voir
            </button>

            <button
              type="button"
              onClick={() => setFilter("active")}
              className={[
                "rounded-full border px-4 py-2 text-sm font-medium transition",
                filter === "active"
                  ? "border-[color:color-mix(in_srgb,var(--cs-account-accent)_22%,white)] bg-[color:color-mix(in_srgb,var(--cs-account-accent)_12%,white)] text-[var(--cs-ink-1)]"
                  : "border-[color:color-mix(in_srgb,var(--cs-line-soft)_68%,white)] bg-white/52 text-[var(--cs-ink-3)]",
              ].join(" ")}
            >
              Actifs seulement
            </button>

            <button
              type="button"
              onClick={() => setFilter("catalog")}
              className={[
                "rounded-full border px-4 py-2 text-sm font-medium transition",
                filter === "catalog"
                  ? "border-[color:color-mix(in_srgb,var(--cs-account-accent)_22%,white)] bg-[color:color-mix(in_srgb,var(--cs-account-accent)_12%,white)] text-[var(--cs-ink-1)]"
                  : "border-[color:color-mix(in_srgb,var(--cs-line-soft)_68%,white)] bg-white/52 text-[var(--cs-ink-3)]",
              ].join(" ")}
            >
              Catalogue CloneStore
            </button>
          </div>
        </div>
      </section>

      {loading ? (
        <section className="cs-card">
          <div className="relative flex items-center gap-3">
            <Loader2 className="h-4 w-4 animate-spin text-[var(--cs-account-accent)]" />
            <p className="text-sm text-[var(--cs-ink-4)]">
              Chargement des employés…
            </p>
          </div>
        </section>
      ) : null}

      {!loading && error ? (
        <section className="cs-card">
          <div className="relative flex items-start gap-3">
            <AlertCircle className="mt-0.5 h-4 w-4 text-[var(--cs-danger)]" />
            <div className="space-y-1">
              <p className="text-sm font-semibold text-[var(--cs-ink-1)]">
                Chargement impossible
              </p>
              <p className="text-sm text-[var(--cs-ink-4)]">{error}</p>
            </div>
          </div>
        </section>
      ) : null}

      {!loading && !error && visibleSlugs.length === 0 ? (
        <section className="cs-card">
          <div className="relative space-y-4 text-center">
            <div className="mx-auto flex h-14 w-14 items-center justify-center rounded-2xl border border-[color:color-mix(in_srgb,var(--cs-line-soft)_68%,white)] bg-white/60 text-[var(--cs-account-accent)] shadow-[var(--cs-shadow-soft)]">
              <Sparkles className="h-5 w-5" />
            </div>

            <div className="space-y-2">
              <p className="text-lg font-semibold text-[var(--cs-ink-1)]">
                Aucun employé à afficher
              </p>
              <p className="mx-auto max-w-xl text-sm text-[var(--cs-ink-4)]">
                Cette vue se remplira dès que des employés seront activés ou
                visibles dans le catalogue retenu.
              </p>
            </div>

            <div className="flex flex-wrap items-center justify-center gap-3">
              <ActionLink href="/agents" label="Voir la boutique" primary />
              <ActionLink href="/questions" label="Ouvrir l’assistant" />
            </div>
          </div>
        </section>
      ) : null}

      {!loading && !error && visibleSlugs.length > 0 ? (
        <section className="grid gap-4 md:grid-cols-2 2xl:grid-cols-3">
          {visibleSlugs.map((slug) => (
            <AgentCard
              key={slug}
              slug={slug}
              order={ordersBySlug.get(slug) ?? null}
            />
          ))}
        </section>
      ) : null}

      <section className="grid gap-4 md:grid-cols-3">
        <div className="cs-card">
          <div className="relative flex items-start gap-3">
            <BadgeCheck className="mt-0.5 h-4 w-4 text-[var(--cs-success)]" />
            <div className="space-y-1">
              <p className="text-sm font-semibold text-[var(--cs-ink-1)]">
                Accès rapides
              </p>
              <p className="text-sm text-[var(--cs-ink-4)]">
                Pierre actif doit être atteignable en un clic.
              </p>
            </div>
          </div>
        </div>

        <div className="cs-card">
          <div className="relative flex items-start gap-3">
            <Settings2 className="mt-0.5 h-4 w-4 text-[var(--cs-account-accent)]" />
            <div className="space-y-1">
              <p className="text-sm font-semibold text-[var(--cs-ink-1)]">
                Configuration
              </p>
              <p className="text-sm text-[var(--cs-ink-4)]">
                L’utilisateur doit comprendre quoi régler et où aller.
              </p>
            </div>
          </div>
        </div>

        <div className="cs-card">
          <div className="relative flex items-start gap-3">
            <Clock3 className="mt-0.5 h-4 w-4 text-[var(--cs-info)]" />
            <div className="space-y-1">
              <p className="text-sm font-semibold text-[var(--cs-ink-1)]">
                Continuité
              </p>
              <p className="text-sm text-[var(--cs-ink-4)]">
                Même les employés à venir doivent déjà exister proprement dans
                l’univers CloneStore.
              </p>
            </div>
          </div>
        </div>
      </section>
    </div>
  );
}
