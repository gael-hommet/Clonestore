"use client";

import Link from "next/link";
import { useMemo, useState } from "react";
import { useRouter } from "next/navigation";
import type { SupabaseClient } from "@supabase/supabase-js";
import {
  ArrowRight,
  Bot,
  KeyRound,
  Loader2,
  Lock,
  Mail,
  ShieldCheck,
  Sparkles,
  UserRound,
  Waypoints,
} from "lucide-react";
import { getSupabase } from "@/lib/supabase";

function ActionButton({
  href,
  label,
  primary = false,
  icon,
}: {
  href: string;
  label: string;
  primary?: boolean;
  icon?: React.ReactNode;
}) {
  return (
    <Link
      href={href}
      className={[
        "inline-flex h-11 items-center justify-center gap-2 rounded-full px-5 text-sm font-medium transition-all duration-[var(--cs-speed-fast)] ease-[var(--cs-ease-premium)]",
        "border",
        primary
          ? [
              "border-[color:color-mix(in_srgb,var(--cs-account-accent)_18%,white)]",
              "bg-[linear-gradient(135deg,color-mix(in_srgb,var(--cs-account-accent)_92%,white),color-mix(in_srgb,var(--cs-accent-blue)_30%,var(--cs-account-accent)))]",
              "text-white",
              "shadow-[0_18px_40px_color-mix(in_srgb,var(--cs-account-accent)_18%,transparent)]",
              "hover:-translate-y-0.5 hover:shadow-[0_24px_52px_color-mix(in_srgb,var(--cs-account-accent)_24%,transparent)]",
            ].join(" ")
          : [
              "border-[color:color-mix(in_srgb,var(--cs-line-soft)_68%,white)]",
              "bg-[linear-gradient(to_bottom,rgba(255,255,255,0.82),rgba(255,250,244,0.56))]",
              "text-[var(--cs-ink-2)]",
              "hover:-translate-y-0.5 hover:bg-white/92",
            ].join(" "),
      ].join(" ")}
    >
      {label}
      {icon}
    </Link>
  );
}

function SectionTitle({
  kicker,
  title,
  text,
}: {
  kicker: string;
  title: string;
  text: string;
}) {
  return (
    <div className="max-w-3xl space-y-3">
      <p className="text-xs font-semibold uppercase tracking-[0.22em] text-[var(--cs-ink-4)]">
        {kicker}
      </p>
      <h2 className="cs-heading text-2xl md:text-3xl">{title}</h2>
      <p className="text-sm text-[var(--cs-ink-4)] md:text-base">{text}</p>
    </div>
  );
}

function InfoCard({
  title,
  text,
  icon,
  tone = "violet",
}: {
  title: string;
  text: string;
  icon: React.ReactNode;
  tone?: "violet" | "blue" | "rose" | "green";
}) {
  const toneClass =
    tone === "violet"
      ? "text-[var(--cs-account-accent)]"
      : tone === "blue"
        ? "text-[var(--cs-info)]"
        : tone === "rose"
          ? "text-[var(--cs-danger)]"
          : "text-[var(--cs-success)]";

  return (
    <div className="cs-card h-full">
      <div className="relative flex h-full flex-col gap-4">
        <div className={`flex items-center gap-2 ${toneClass}`}>
          {icon}
          <span className="text-sm font-medium text-[var(--cs-ink-2)]">
            {title}
          </span>
        </div>
        <p className="text-sm text-[var(--cs-ink-4)]">{text}</p>
      </div>
    </div>
  );
}

function Field({
  label,
  icon,
  type = "text",
  value,
  onChange,
  placeholder,
  autoComplete,
}: {
  label: string;
  icon: React.ReactNode;
  type?: string;
  value: string;
  onChange: (value: string) => void;
  placeholder: string;
  autoComplete?: string;
}) {
  return (
    <label className="block space-y-2">
      <span className="text-sm font-medium text-[var(--cs-ink-2)]">{label}</span>
      <div className="flex items-center gap-3 rounded-[1.2rem] border border-[color:color-mix(in_srgb,var(--cs-line-soft)_68%,white)] bg-[linear-gradient(to_bottom,rgba(255,255,255,0.78),rgba(255,250,244,0.52))] px-4 py-3 shadow-[var(--cs-shadow-soft)]">
        <span className="text-[var(--cs-account-accent)]">{icon}</span>
        <input
          type={type}
          value={value}
          onChange={(e) => onChange(e.target.value)}
          placeholder={placeholder}
          autoComplete={autoComplete}
          className="w-full border-0 bg-transparent text-sm text-[var(--cs-ink-2)] outline-none placeholder:text-[var(--cs-ink-4)]"
        />
      </div>
    </label>
  );
}

export default function LoginPage() {
  const router = useRouter();
  const supabase = useMemo(() => getSupabase() as SupabaseClient | null, []);

  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");

  const [mode, setMode] = useState<"password" | "magic">("password");
  const [isLoading, setIsLoading] = useState(false);
  const [success, setSuccess] = useState<string | null>(null);
  const [error, setError] = useState<string | null>(null);

  const canSubmit = useMemo(() => {
    if (!email.trim()) return false;
    if (mode === "password" && !password.trim()) return false;
    return !isLoading;
  }, [email, password, mode, isLoading]);

  async function handlePasswordLogin() {
    if (!supabase) {
      setError(
        "Configuration Supabase manquante. Vérifie tes variables d’environnement puis redémarre le projet.",
      );
      return;
    }

    setIsLoading(true);
    setError(null);
    setSuccess(null);

    try {
      const { error: signInError } = await supabase.auth.signInWithPassword({
        email: email.trim(),
        password,
      });

      if (signInError) {
        throw signInError;
      }

      router.push("/profile");
      router.refresh();
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "Impossible de se connecter pour le moment.",
      );
    } finally {
      setIsLoading(false);
    }
  }

  async function handleMagicLink() {
    if (!supabase) {
      setError(
        "Configuration Supabase manquante. Vérifie tes variables d’environnement puis redémarre le projet.",
      );
      return;
    }

    setIsLoading(true);
    setError(null);
    setSuccess(null);

    try {
      const redirectTo =
        typeof window !== "undefined"
          ? `${window.location.origin}/profile`
          : undefined;

      const { error: otpError } = await supabase.auth.signInWithOtp({
        email: email.trim(),
        options: {
          emailRedirectTo: redirectTo,
        },
      });

      if (otpError) {
        throw otpError;
      }

      setSuccess(
        "Lien envoyé. Ouvre ton email puis reviens dans CloneStore depuis le lien de connexion.",
      );
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "Impossible d’envoyer le lien magique pour le moment.",
      );
    } finally {
      setIsLoading(false);
    }
  }

  async function handleSubmit(e: React.FormEvent) {
    e.preventDefault();
    if (!canSubmit) return;

    if (mode === "password") {
      await handlePasswordLogin();
      return;
    }

    await handleMagicLink();
  }

  return (
    <div className="cs-page-shell py-10 md:py-14">
      <div className="space-y-6">
        <section className="cs-command-surface overflow-hidden">
          <div className="cs-system-halo" />

          <div className="relative grid gap-6 xl:grid-cols-[1.02fr_0.98fr] xl:items-start">
            <div className="space-y-6">
              <div className="flex flex-wrap items-center gap-2">
                <span className="cs-pill">
                  <UserRound className="h-3.5 w-3.5 text-[var(--cs-account-accent)]" />
                  <span>Connexion CloneStore</span>
                </span>
                <span className="cs-pill">
                  <ShieldCheck className="h-3.5 w-3.5 text-[var(--cs-success)]" />
                  <span>Entrée vers le cockpit</span>
                </span>
              </div>

              <div className="space-y-4">
                <h1 className="cs-heading text-3xl md:text-5xl">
                  Entrez dans CloneStore
                  <br />
                  <span className="cs-gradient-text">
                    sans friction, sans confusion.
                  </span>
                </h1>

                <p className="max-w-3xl text-sm text-[var(--cs-ink-4)] md:text-base">
                  Une fois connecté, vous arrivez directement dans votre cockpit :
                  employés IA, messages, missions, traçabilité, réglages,
                  personnalisation et contrôle total du système.
                </p>
              </div>

              <div className="flex flex-wrap gap-3">
                <ActionButton
                  href="/signup"
                  label="Créer un compte"
                  primary
                  icon={<ArrowRight className="h-4 w-4" />}
                />
                <ActionButton
                  href="/questions"
                  label="Poser une question"
                  icon={<Bot className="h-4 w-4" />}
                />
                <ActionButton
                  href="/agents"
                  label="Voir la boutique"
                  icon={<Sparkles className="h-4 w-4" />}
                />
              </div>

              <div className="grid gap-4 sm:grid-cols-2">
                <InfoCard
                  title="Entrée cockpit"
                  text="Après connexion, le client doit entrer là où il agit vraiment : son espace CloneStore."
                  icon={<Waypoints className="h-4 w-4" />}
                  tone="violet"
                />
                <InfoCard
                  title="Simplicité"
                  text="L’expérience doit rester limpide, même pour quelqu’un qui n’aime pas l’informatique."
                  icon={<Sparkles className="h-4 w-4" />}
                  tone="green"
                />
                <InfoCard
                  title="Confiance"
                  text="Validation, gouvernance, mémoire, trace et sécurité sont au cœur du système."
                  icon={<Lock className="h-4 w-4" />}
                  tone="blue"
                />
                <InfoCard
                  title="Assistance"
                  text="L’assistant CloneStore reste accessible avant, pendant et après la connexion."
                  icon={<Bot className="h-4 w-4" />}
                  tone="rose"
                />
              </div>
            </div>

            <div className="cs-panel">
              <div className="relative space-y-6">
                <div className="space-y-2">
                  <p className="text-xs uppercase tracking-[0.18em] text-[var(--cs-ink-4)]">
                    Accès sécurisé
                  </p>
                  <h2 className="cs-heading text-2xl">Se connecter</h2>
                  <p className="text-sm text-[var(--cs-ink-4)]">
                    Choisis la méthode la plus simple pour toi.
                  </p>
                </div>

                <div className="flex flex-wrap gap-2">
                  <button
                    type="button"
                    onClick={() => {
                      setMode("password");
                      setError(null);
                      setSuccess(null);
                    }}
                    className={[
                      "rounded-full border px-4 py-2 text-sm font-medium transition",
                      mode === "password"
                        ? "border-[color:color-mix(in_srgb,var(--cs-account-accent)_22%,white)] bg-[color:color-mix(in_srgb,var(--cs-account-accent)_12%,white)] text-[var(--cs-ink-1)]"
                        : "border-[color:color-mix(in_srgb,var(--cs-line-soft)_68%,white)] bg-white/52 text-[var(--cs-ink-3)]",
                    ].join(" ")}
                  >
                    Email + mot de passe
                  </button>

                  <button
                    type="button"
                    onClick={() => {
                      setMode("magic");
                      setError(null);
                      setSuccess(null);
                    }}
                    className={[
                      "rounded-full border px-4 py-2 text-sm font-medium transition",
                      mode === "magic"
                        ? "border-[color:color-mix(in_srgb,var(--cs-account-accent)_22%,white)] bg-[color:color-mix(in_srgb,var(--cs-account-accent)_12%,white)] text-[var(--cs-ink-1)]"
                        : "border-[color:color-mix(in_srgb,var(--cs-line-soft)_68%,white)] bg-white/52 text-[var(--cs-ink-3)]",
                    ].join(" ")}
                  >
                    Lien magique
                  </button>
                </div>

                <form onSubmit={handleSubmit} className="space-y-4">
                  <Field
                    label="Adresse email"
                    icon={<Mail className="h-4 w-4" />}
                    value={email}
                    onChange={setEmail}
                    placeholder="vous@entreprise.com"
                    autoComplete="email"
                    type="email"
                  />

                  {mode === "password" ? (
                    <Field
                      label="Mot de passe"
                      icon={<KeyRound className="h-4 w-4" />}
                      value={password}
                      onChange={setPassword}
                      placeholder="Votre mot de passe"
                      autoComplete="current-password"
                      type="password"
                    />
                  ) : (
                    <div className="cs-card">
                      <div className="relative space-y-2">
                        <p className="text-sm font-medium text-[var(--cs-ink-2)]">
                          Connexion sans mot de passe
                        </p>
                        <p className="text-sm text-[var(--cs-ink-4)]">
                          Nous t’enverrons un lien de connexion directement par
                          email pour entrer dans CloneStore.
                        </p>
                      </div>
                    </div>
                  )}

                  {error ? (
                    <div className="rounded-[1.2rem] border border-[color:rgba(161,75,83,0.22)] bg-[rgba(161,75,83,0.08)] px-4 py-3 text-sm text-[var(--cs-danger)]">
                      {error}
                    </div>
                  ) : null}

                  {success ? (
                    <div className="rounded-[1.2rem] border border-[color:rgba(35,122,82,0.22)] bg-[rgba(35,122,82,0.08)] px-4 py-3 text-sm text-[var(--cs-success)]">
                      {success}
                    </div>
                  ) : null}

                  <button
                    type="submit"
                    disabled={!canSubmit}
                    className={[
                      "inline-flex h-12 w-full items-center justify-center gap-2 rounded-full px-5 text-sm font-medium transition-all duration-[var(--cs-speed-fast)] ease-[var(--cs-ease-premium)]",
                      "border",
                      canSubmit
                        ? "border-[color:color-mix(in_srgb,var(--cs-account-accent)_18%,white)] bg-[linear-gradient(135deg,color-mix(in_srgb,var(--cs-account-accent)_92%,white),color-mix(in_srgb,var(--cs-accent-blue)_30%,var(--cs-account-accent)))] text-white shadow-[0_18px_40px_color-mix(in_srgb,var(--cs-account-accent)_18%,transparent)] hover:-translate-y-0.5"
                        : "border-[color:color-mix(in_srgb,var(--cs-line-soft)_68%,white)] bg-[linear-gradient(to_bottom,rgba(255,255,255,0.62),rgba(255,249,242,0.42))] text-[var(--cs-ink-4)] opacity-70",
                    ].join(" ")}
                  >
                    {isLoading ? (
                      <>
                        <Loader2 className="h-4 w-4 animate-spin" />
                        <span>Connexion…</span>
                      </>
                    ) : (
                      <>
                        <span>
                          {mode === "password"
                            ? "Entrer dans CloneStore"
                            : "Envoyer le lien magique"}
                        </span>
                        <ArrowRight className="h-4 w-4" />
                      </>
                    )}
                  </button>
                </form>

                <div className="flex flex-wrap items-center justify-between gap-3 border-t border-[color:var(--cs-line-soft)] pt-4">
                  <p className="text-sm text-[var(--cs-ink-4)]">
                    Pas encore de compte ?
                  </p>
                  <Link
                    href="/signup"
                    className="inline-flex items-center gap-2 text-sm font-medium text-[var(--cs-ink-2)] transition hover:text-[var(--cs-ink-1)]"
                  >
                    Créer un compte
                    <ArrowRight className="h-4 w-4" />
                  </Link>
                </div>
              </div>
            </div>
          </div>
        </section>

        <section className="grid gap-6 xl:grid-cols-[1fr_1fr]">
          <div className="cs-panel">
            <SectionTitle
              kicker="Pourquoi cette page compte"
              title="La connexion ne doit pas casser l’image premium du produit."
              text="Si l’entrée compte est banale ou confuse, on casse une partie de la promesse CloneStore. Cette page doit rester fluide, simple, rassurante et cohérente."
            />

            <div className="mt-6 grid gap-3">
              <InfoCard
                title="Simplicité d’accès"
                text="Un client doit savoir quoi faire immédiatement, sans jargon et sans friction."
                icon={<UserRound className="h-4 w-4" />}
                tone="violet"
              />
              <InfoCard
                title="Orientation claire"
                text="On n’entre pas dans un profil vide : on entre dans un cockpit vivant."
                icon={<Waypoints className="h-4 w-4" />}
                tone="blue"
              />
              <InfoCard
                title="Confiance"
                text="Le ton, la sécurité et la clarté doivent être perceptibles dès la connexion."
                icon={<ShieldCheck className="h-4 w-4" />}
                tone="green"
              />
            </div>
          </div>

          <div className="cs-panel">
            <SectionTitle
              kicker="Assistance"
              title="Même ici, l’assistant doit pouvoir aider."
              text="Quelqu’un peut bloquer au moment de se connecter. Le support et l’assistant doivent déjà être accessibles à cette étape."
            />

            <div className="mt-6 grid gap-3">
              <InfoCard
                title="Question produit"
                text="Comprendre ce qu’on va trouver une fois connecté."
                icon={<Bot className="h-4 w-4" />}
                tone="violet"
              />
              <InfoCard
                title="Question compte"
                text="Être guidé si on ne sait pas quelle méthode choisir."
                icon={<Mail className="h-4 w-4" />}
                tone="blue"
              />
              <InfoCard
                title="Question cockpit"
                text="Savoir où l’on va arriver et pourquoi c’est le bon point d’entrée."
                icon={<Waypoints className="h-4 w-4" />}
                tone="rose"
              />
            </div>

            <div className="mt-6">
              <ActionButton
                href="/questions"
                label="Ouvrir l’assistant"
                primary
                icon={<ArrowRight className="h-4 w-4" />}
              />
            </div>
          </div>
        </section>
      </div>
    </div>
  );
}
