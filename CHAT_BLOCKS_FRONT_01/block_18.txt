"use client";

import Link from "next/link";
import { usePathname } from "next/navigation";
import {
  Bot,
  BriefcaseBusiness,
  Settings2,
  Sparkles,
  Waypoints,
} from "lucide-react";

function NavItem({
  href,
  label,
  icon,
  active,
}: {
  href: string;
  label: string;
  icon: React.ReactNode;
  active: boolean;
}) {
  return (
    <Link
      href={href}
      className={[
        "inline-flex h-10 items-center gap-2 rounded-full px-4 text-sm font-medium transition-all",
        active
          ? "bg-[rgba(111,70,255,0.1)] text-[color:var(--cs-violet)] shadow-[inset_0_0_0_1px_rgba(111,70,255,0.1)]"
          : "text-[color:var(--cs-ink-3)] hover:bg-white/60 hover:text-[color:var(--cs-ink-1)]",
      ].join(" ")}
    >
      {icon}
      <span>{label}</span>
    </Link>
  );
}

export default function ProfileLayout({
  children,
}: {
  children: React.ReactNode;
}) {
  const pathname = usePathname();

  return (
    <div className="space-y-4">
      <section className="cs-panel rounded-[1.8rem] p-3">
        <div className="flex flex-wrap items-center gap-2">
          <NavItem
            href="/profile"
            label="Cockpit"
            icon={<Waypoints className="h-4 w-4" />}
            active={pathname === "/profile"}
          />
          <NavItem
            href="/profile/agents"
            label="Employés"
            icon={<BriefcaseBusiness className="h-4 w-4" />}
            active={pathname === "/profile/agents"}
          />
          <NavItem
            href="/assistant"
            label="Assistant"
            icon={<Bot className="h-4 w-4" />}
            active={pathname === "/assistant"}
          />
          <NavItem
            href="/agents"
            label="Boutique"
            icon={<Sparkles className="h-4 w-4" />}
            active={pathname === "/agents"}
          />
          <NavItem
            href="/agents/pierre/setup"
            label="Configuration"
            icon={<Settings2 className="h-4 w-4" />}
            active={pathname === "/agents/pierre/setup"}
          />
        </div>
      </section>

      {children}
    </div>
  );
}
