  ArrowRight,
  Bot,
  BriefcaseBusiness,
  Building2,
  Loader2,
  Lock,
  Mail,
  ShieldCheck,
  Sparkles,
  UserRound,
  Waypoints,
} from "lucide-react";
import { getSupabase } from "@/lib/supabase";

function ActionButton({
  href,
  label,
  primary = false,
  icon,
}: {
  href: string;
  label: string;
  primary?: boolean;
  icon?: React.ReactNode;
}) {
  return (
    <Link
      href={href}
      className={[
        "inline-flex h-11 items-center justify-center gap-2 rounded-full px-5 text-sm font-medium transition-all duration-[var(--cs-speed-fast)] ease-[var(--cs-ease-premium)]",
        "border",
        primary
          ? [
              "border-[color:color-mix(in_srgb,var(--cs-account-accent)_18%,white)]",
              "bg-[linear-gradient(135deg,color-mix(in_srgb,var(--cs-account-accent)_92%,white),color-mix(in_srgb,var(--cs-accent-blue)_30%,var(--cs-account-accent)))]",
              "text-white",
              "shadow-[0_18px_40px_color-mix(in_srgb,var(--cs-account-accent)_18%,transparent)]",
              "hover:-translate-y-0.5 hover:shadow-[0_24px_52px_color-mix(in_srgb,var(--cs-account-accent)_24%,transparent)]",
            ].join(" ")
          : [
              "border-[color:color-mix(in_srgb,var(--cs-line-soft)_68%,white)]",
              "bg-[linear-gradient(to_bottom,rgba(255,255,255,0.82),rgba(255,250,244,0.56))]",
              "text-[var(--cs-ink-2)]",
              "hover:-translate-y-0.5 hover:bg-white/92",
            ].join(" "),
      ].join(" ")}
    >
      {label}
      {icon}
    </Link>
  );
}

function SectionTitle({
  kicker,
  title,
  text,
}: {
  kicker: string;
  title: string;
  text: string;
}) {
  return (
    <div className="max-w-3xl space-y-3">
      <p className="text-xs font-semibold uppercase tracking-[0.22em] text-[var(--cs-ink-4)]">
        {kicker}
      </p>
      <h2 className="cs-heading text-2xl md:text-3xl">{title}</h2>
      <p className="text-sm text-[var(--cs-ink-4)] md:text-base">{text}</p>
    </div>
  );
}

function InfoCard({
  title,
  text,
  icon,
  tone = "violet",
}: {
  title: string;
  text: string;
  icon: React.ReactNode;
  tone?: "violet" | "blue" | "rose" | "green";
}) {
  const toneClass =
    tone === "violet"
      ? "text-[var(--cs-account-accent)]"
      : tone === "blue"
        ? "text-[var(--cs-info)]"
        : tone === "rose"
          ? "text-[var(--cs-danger)]"
          : "text-[var(--cs-success)]";

  return (
    <div className="cs-card h-full">
      <div className="relative flex h-full flex-col gap-4">
        <div className={`flex items-center gap-2 ${toneClass}`}>
          {icon}
          <span className="text-sm font-medium text-[var(--cs-ink-2)]">
            {title}
          </span>
        </div>
        <p className="text-sm text-[var(--cs-ink-4)]">{text}</p>
      </div>
    </div>
  );
}

function Field({
  label,
  icon,
  type = "text",
  value,
  onChange,
  placeholder,
  autoComplete,
}: {
  label: string;
  icon: React.ReactNode;
  type?: string;
  value: string;
  onChange: (value: string) => void;
  placeholder: string;
  autoComplete?: string;
}) {
  return (
    <label className="block space-y-2">
      <span className="text-sm font-medium text-[var(--cs-ink-2)]">{label}</span>
      <div className="flex items-center gap-3 rounded-[1.2rem] border border-[color:color-mix(in_srgb,var(--cs-line-soft)_68%,white)] bg-[linear-gradient(to_bottom,rgba(255,255,255,0.78),rgba(255,250,244,0.52))] px-4 py-3 shadow-[var(--cs-shadow-soft)]">
        <span className="text-[var(--cs-account-accent)]">{icon}</span>
        <input
          type={type}
          value={value}
          onChange={(e) => onChange(e.target.value)}
          placeholder={placeholder}
          autoComplete={autoComplete}
          className="w-full border-0 bg-transparent text-sm text-[var(--cs-ink-2)] outline-none placeholder:text-[var(--cs-ink-4)]"
        />
      </div>
    </label>
  );
}

export default function SignupPage() {
  const router = useRouter();
  const supabase = useMemo(() => getSupabase() as SupabaseClient | null, []);

  const [fullName, setFullName] = useState("");
  const [companyName, setCompanyName] = useState("");
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [confirmPassword, setConfirmPassword] = useState("");

  const [isLoading, setIsLoading] = useState(false);
  const [success, setSuccess] = useState<string | null>(null);
  const [error, setError] = useState<string | null>(null);

  const passwordChecks = useMemo(
    () => ({
      length: password.length >= 8,
      upper: /[A-ZÀ-ÖØ-Ý]/.test(password),
      lower: /[a-zà-öø-ÿ]/.test(password),
      number: /\d/.test(password),
      match: password.length > 0 && password === confirmPassword,
    }),
    [password, confirmPassword],
  );

  const canSubmit = useMemo(() => {
    return (
      fullName.trim().length > 0 &&
      email.trim().length > 0 &&
      passwordChecks.length &&
      passwordChecks.upper &&
      passwordChecks.lower &&
      passwordChecks.number &&
      passwordChecks.match &&
      !isLoading
    );
  }, [fullName, email, passwordChecks, isLoading]);

  async function handleSignup(e: React.FormEvent) {
    e.preventDefault();
    if (!canSubmit) return;

    if (!supabase) {
      setError(
        "Configuration Supabase manquante. Vérifie tes variables d’environnement puis redémarre le projet.",
      );
      return;
    }

    setIsLoading(true);
    setError(null);
    setSuccess(null);

    try {
      const redirectTo =
        typeof window !== "undefined"
          ? `${window.location.origin}/profile`
          : undefined;

      const { data, error: signUpError } = await supabase.auth.signUp({
        email: email.trim(),
        password,
        options: {
          emailRedirectTo: redirectTo,
          data: {
            full_name: fullName.trim(),
            company_name: companyName.trim() || null,
          },
        },
      });

      if (signUpError) {
        throw signUpError;
      }

      if (data.session) {
        router.push("/profile");
        router.refresh();
        return;
      }

      setSuccess(
        "Compte créé. Vérifie maintenant ton email pour confirmer ton accès puis entre dans CloneStore.",
      );
    } catch (err) {
      setError(
        err instanceof Error
          ? err.message
          : "Impossible de créer le compte pour le moment.",
      );
    } finally {
      setIsLoading(false);
    }
  }

  function CheckItem({
    ok,
    text,
  }: {
    ok: boolean;
    text: string;
  }) {
    return (
      <div className="flex items-center gap-2 text-xs">
        <span
          className={[
            "h-2.5 w-2.5 rounded-full",
            ok ? "bg-[var(--cs-success)]" : "bg-[rgba(20,20,22,0.16)]",
          ].join(" ")}
        />
        <span className={ok ? "text-[var(--cs-ink-2)]" : "text-[var(--cs-ink-4)]"}>
          {text}
        </span>
      </div>
    );
  }

  return (
    <div className="cs-page-shell py-10 md:py-14">
      <div className="space-y-6">
        <section className="cs-command-surface overflow-hidden">
          <div className="cs-system-halo" />

          <div className="relative grid gap-6 xl:grid-cols-[1.02fr_0.98fr] xl:items-start">
            <div className="space-y-6">
              <div className="flex flex-wrap items-center gap-2">
                <span className="cs-pill">
                  <Sparkles className="h-3.5 w-3.5 text-[var(--cs-account-accent)]" />
                  <span>Création de compte CloneStore</span>
                </span>
                <span className="cs-pill">
                  <ShieldCheck className="h-3.5 w-3.5 text-[var(--cs-success)]" />
                  <span>Première entrée dans le système</span>
                </span>
              </div>

              <div className="space-y-4">
                <h1 className="cs-heading text-3xl md:text-5xl">
                  Créez votre compte
                  <br />
                  <span className="cs-gradient-text">
                    et entrez dans votre futur cockpit.
                  </span>
                </h1>

                <p className="max-w-3xl text-sm text-[var(--cs-ink-4)] md:text-base">
                  L’inscription doit rester simple, propre et rassurante. On entre
                  ensuite dans CloneStore pour configurer l’entreprise, activer les
                  bons employés IA et piloter le système depuis un seul endroit.
                </p>
              </div>

              <div className="flex flex-wrap gap-3">
                <ActionButton
                  href="/login"
                  label="Déjà un compte ? Se connecter"
                  primary
                  icon={<ArrowRight className="h-4 w-4" />}
                />
                <ActionButton
                  href="/questions"
                  label="Poser une question"
                  icon={<Bot className="h-4 w-4" />}
                />
                <ActionButton
                  href="/agents"
                  label="Voir la boutique"
                  icon={<BriefcaseBusiness className="h-4 w-4" />}
                />
              </div>

              <div className="grid gap-4 sm:grid-cols-2">
                <InfoCard
                  title="Entrée propre"
                  text="Créer un compte doit être fluide, rapide et sans intimidation."
                  icon={<UserRound className="h-4 w-4" />}
                  tone="violet"
                />
                <InfoCard
                  title="Suite logique"
                  text="Après création, le client entre dans un cockpit clair plutôt que dans un espace vide."
                  icon={<Waypoints className="h-4 w-4" />}
                  tone="blue"
                />
                <InfoCard
                  title="Sécurité"
                  text="La confiance se construit aussi dans la qualité perçue de l’accès et du cadre produit."
                  icon={<Lock className="h-4 w-4" />}
                  tone="green"
                />
                <InfoCard
                  title="Assistance"
                  text="L’assistant CloneStore reste disponible si quelque chose bloque ou si une question subsiste."
                  icon={<Bot className="h-4 w-4" />}
                  tone="rose"
                />
              </div>
            </div>

            <div className="cs-panel">
              <div className="relative space-y-6">
                <div className="space-y-2">
                  <p className="text-xs uppercase tracking-[0.18em] text-[var(--cs-ink-4)]">
                    Nouveau compte
                  </p>
                  <h2 className="cs-heading text-2xl">Créer un compte</h2>
                  <p className="text-sm text-[var(--cs-ink-4)]">
                    Quelques informations simples pour démarrer proprement.
                  </p>
                </div>

                <form onSubmit={handleSignup} className="space-y-4">
                  <Field
                    label="Nom complet"
                    icon={<UserRound className="h-4 w-4" />}
                    value={fullName}
                    onChange={setFullName}
                    placeholder="Prénom Nom"
                    autoComplete="name"
                  />

                  <Field
                    label="Entreprise (optionnel)"
                    icon={<Building2 className="h-4 w-4" />}
                    value={companyName}
                    onChange={setCompanyName}
                    placeholder="Nom de votre entreprise"
                    autoComplete="organization"
                  />

                  <Field
                    label="Adresse email"
                    icon={<Mail className="h-4 w-4" />}
                    value={email}
                    onChange={setEmail}
                    placeholder="vous@entreprise.com"
                    autoComplete="email"
                    type="email"
                  />

                  <Field
                    label="Mot de passe"
                    icon={<Lock className="h-4 w-4" />}
                    value={password}
                    onChange={setPassword}
                    placeholder="Choisissez un mot de passe"
                    autoComplete="new-password"
                    type="password"
                  />

                  <Field
                    label="Confirmer le mot de passe"
                    icon={<Lock className="h-4 w-4" />}
                    value={confirmPassword}
                    onChange={setConfirmPassword}
                    placeholder="Retapez le mot de passe"
                    autoComplete="new-password"
                    type="password"
                  />

                  <div className="cs-card">
                    <div className="relative space-y-3">
                      <p className="text-sm font-medium text-[var(--cs-ink-2)]">
                        Vérifications mot de passe
                      </p>
                      <div className="grid gap-2 sm:grid-cols-2">
                        <CheckItem ok={passwordChecks.length} text="Au moins 8 caractères" />
                        <CheckItem ok={passwordChecks.upper} text="Une majuscule" />
                        <CheckItem ok={passwordChecks.lower} text="Une minuscule" />
                        <CheckItem ok={passwordChecks.number} text="Un chiffre" />
                        <CheckItem ok={passwordChecks.match} text="Confirmation identique" />
                      </div>
                    </div>
                  </div>

                  {error ? (
                    <div className="rounded-[1.2rem] border border-[color:rgba(161,75,83,0.22)] bg-[rgba(161,75,83,0.08)] px-4 py-3 text-sm text-[var(--cs-danger)]">
                      {error}
                    </div>
                  ) : null}

                  {success ? (
                    <div className="rounded-[1.2rem] border border-[color:rgba(35,122,82,0.22)] bg-[rgba(35,122,82,0.08)] px-4 py-3 text-sm text-[var(--cs-success)]">
                      {success}
                    </div>
                  ) : null}

                  <button
                    type="submit"
                    disabled={!canSubmit}
                    className={[
                      "inline-flex h-12 w-full items-center justify-center gap-2 rounded-full px-5 text-sm font-medium transition-all duration-[var(--cs-speed-fast)] ease-[var(--cs-ease-premium)]",
                      "border",
                      canSubmit
                        ? "border-[color:color-mix(in_srgb,var(--cs-account-accent)_18%,white)] bg-[linear-gradient(135deg,color-mix(in_srgb,var(--cs-account-accent)_92%,white),color-mix(in_srgb,var(--cs-accent-blue)_30%,var(--cs-account-accent)))] text-white shadow-[0_18px_40px_color-mix(in_srgb,var(--cs-account-accent)_18%,transparent)] hover:-translate-y-0.5"
                        : "border-[color:color-mix(in_srgb,var(--cs-line-soft)_68%,white)] bg-[linear-gradient(to_bottom,rgba(255,255,255,0.62),rgba(255,249,242,0.42))] text-[var(--cs-ink-4)] opacity-70",
                    ].join(" ")}
                  >
                    {isLoading ? (
                      <>
                        <Loader2 className="h-4 w-4 animate-spin" />
                        <span>Création du compte…</span>
                      </>
                    ) : (
                      <>
                        <span>Créer mon compte CloneStore</span>
                        <ArrowRight className="h-4 w-4" />
                      </>
                    )}
                  </button>
                </form>

                <div className="flex flex-wrap items-center justify-between gap-3 border-t border-[color:var(--cs-line-soft)] pt-4">
                  <p className="text-sm text-[var(--cs-ink-4)]">
                    Vous avez déjà un compte ?
                  </p>
                  <Link
                    href="/login"
                    className="inline-flex items-center gap-2 text-sm font-medium text-[var(--cs-ink-2)] transition hover:text-[var(--cs-ink-1)]"
                  >
                    Se connecter
                    <ArrowRight className="h-4 w-4" />
                  </Link>
                </div>
              </div>
            </div>
          </div>
        </section>

        <section className="grid gap-6 xl:grid-cols-[1fr_1fr]">
          <div className="cs-panel">
            <SectionTitle
              kicker="Pourquoi cette page compte"
              title="L’inscription prépare la suite de l’expérience."
              text="Créer un compte ne doit pas être une formalité froide. C’est la première vraie entrée dans le système CloneStore."
            />

            <div className="mt-6 grid gap-3">
              <InfoCard
